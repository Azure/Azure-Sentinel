// Cyble Threat Intelligence parser for External_SourceRatings
//
// Replace SourceSystem value with SourceSystem of your TAXII connector
let SourceSystemName = "";
let Base =
    ThreatIntelIndicators
    | where isempty(SourceSystemName) or SourceSystem =~ SourceSystemName
    | extend DataJson = todynamic(tostring(Data))
    | extend RefCount = coalesce(array_length(DataJson.external_references), 0);
let WithRefs =
    Base
    | where RefCount > 0
    | mv-apply ExtRef = DataJson.external_references on (
          extend SourceName = tostring(ExtRef.source_name),
                 RefDescription = tostring(ExtRef.description)
        | summarize ParsedFields = make_bag(bag_pack(
                        strcat(tolower(SourceName), ".risk_score"),
                            toint(trim(@"\s*", extract(@"risk_score\s*=\s*([^;]*)", 1, RefDescription))),
                        strcat(tolower(SourceName), ".confidence_rating"),
                            trim(@"\s*", extract(@"confidence_rating\s*=\s*([^;]*)", 1, RefDescription)))),
                    SourceNames = make_set(SourceName)
      )
    | extend Data = bag_merge(
                        DataJson,
                        ParsedFields,
                        bag_pack("ext_source_name", strcat_array(SourceNames, ", ")))
    | project-away ParsedFields, SourceNames;
let WithoutRefs =
    Base
    | where RefCount == 0
    | extend Data = DataJson;
union WithRefs, WithoutRefs
| project-away DataJson, RefCount
